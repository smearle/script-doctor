% ID/OOD teaser figure for the rule-conditioned NCA WM paper.
%
% Single-document standalone TikZ figure. The dynamics block (embed →
% banded NCA body → readout) is copied from ../architecture/nca_wm_mini.tex
% and placed once per regime (in-distribution training and OOD eval),
% with PuzzleScript engine frames projected onto small Box pics around
% it. Each row also has a small "clear" Box just before the s_0 frame
% containing a miniaturized snippet of the game's PuzzleScript rules,
% representing the rule-conditioning input the NCA-WM gets alongside s_0.
%
% Forward (blue/red), loss (dark orange), and autoregressive (dashed
% red) connections are all drawn natively in TikZ — no external PNG
% mini-diagram is included.
%
% Required engine renders (produced by nca_wm/scripts/dump_engine_frames.py
% and nca_wm/scripts/dump_ood_rollout_frames.py):
%   rollouts/<game>/engine_t0.png      (ID s_0)
%   rollouts/<game>/engine_t1.png      (ID s_1)
%   rollouts/<game>/pred_t{0..3}.png   (OOD s_0 + autoregressive ŝ)
%
% Build:
%   cd nca_wm/paper/figures/id_ood_teaser
%   pdflatex teaser.tex
%

% --- PS frame: small Box with image projected on east face. ---
\newcommand{\PSframe}[4]{%
  \pic[shift={(#2, #3, 0)}] at (0,0,0)
      {Box={name=#1, caption=, xlabel={{,}}, zlabel={},
            fill=PSFrameFill, height=4, width=1.0, depth=4}};%
  \node[canvas is zy plane at x={#2 + 0.2}] at (0, #3)
      {\includegraphics[width=0.8cm,height=0.8cm]{#4}};%
}

% --- Code Box: clear (white) Box with a tiny PS rule snippet on the
%     east face. Slightly deeper than PS frames so wide rule lines fit.
%     #1 = pic name, #2 = x_shift, #3 = y_shift, #4 = code text
\newcommand{\Codebox}[4]{%
  \pic[shift={(#2, #3, 0)}] at (0,0,0)
      {Box={name=#1, caption=, xlabel={{,}}, zlabel={},
            fill=CodeFill, opacity=0.6,
            height=4, width=1.0, depth=5}};%
  \node[canvas is zy plane at x={#2 + 0.2}, font=\tiny\ttfamily,
        align=left, inner sep=0.6pt, text width=1cm] at (0, #3)
      {#4};%
}

% --- NCA-WM block (embed -> 3 NCA bands -> readout) + faint loop arrow.
\newcommand{\NCAblock}[3]{%
  \pic[shift={({#2 - 1.4}, #3, 0)}] at (0,0,0)
      {Box={name=embed_#1, caption=, xlabel={{,}}, zlabel={},
            fill=EmbedColor, height=8, width=1.5, depth=8}};
  \pic[shift={({#2 + 0.0}, #3, 0)}] at (0,0,0)
      {RightBandedBox={name=nca_#1, caption=, xlabel={{,,}}, zlabel={},
            fill=NCAColor, bandfill=NCABandColor,
            height=8, width={1.5,1.5,1.5}, depth=8}};
  \pic[shift={({#2 + 1.7}, #3, 0)}] at (0,0,0)
      {Box={name=readout_#1, caption=, xlabel={{,}}, zlabel={},
            fill=ReadoutColor, height=8, width=1.5, depth=8}};
  \draw[connection, -Stealth] (embed_#1-east) -- (nca_#1-west);
  \draw[connection, -Stealth] (nca_#1-east)   -- (readout_#1-west);
  \path let \p1 = (nca_#1-east), \p2 = (nca_#1-west) in
    coordinate (loopOutR_#1)  at (\x1+0.25cm, \y1)
    coordinate (loopOutRD_#1) at (\x1+0.25cm, \y1-1.0cm)
    coordinate (loopInLD_#1)  at (\x2-0.35cm, \y2-1.0cm)
    coordinate (loopInL_#1)   at (\x2-0.35cm, \y2);
  \draw[-Stealth, line width=0.4mm, opacity=0.32, rounded corners=4pt]
      (nca_#1-east) -- (loopOutR_#1) -- (loopOutRD_#1)
      -- (loopInLD_#1) -- (loopInL_#1) -- (nca_#1-west);
}
