%   rollouts/<game>/engine_t1.png      (ID s_1)
%   rollouts/<game>/pred_t{0..3}.png   (OOD s_0 + autoregressive ŝ)
%
% Build:
%   cd nca_wm/paper/figures/id_ood_teaser
%   pdflatex teaser.tex
%

% --- helper macros (no \foreach) ---
% Place a small PS-frame Box and project an image on its east face.
%   #1 = pic name, #2 = x_shift, #3 = y_shift, #4 = image path
\newcommand{\PSframe}[4]{%
  \pic[shift={(#2, #3, 0)}] at (0,0,0)
      {Box={name=#1, caption=, xlabel={{,}}, zlabel={},
            fill=PSFrameFill, height=6, width=1.5, depth=6}};%
  % east face is at x = #2 + 1.5*0.2 = #2 + 0.3
  \node[canvas is zy plane at x={#2 + 0.3}] at (0, #3)
      {\includegraphics[width=1.2cm,height=1.2cm]{#4}};%
}

% Place an NCA-WM dynamics block (embed -> 3 NCA bands -> readout) plus
% a faint loop arrow underneath.
%   #1 = name suffix (id / ood), #2 = y_center
\newcommand{\NCAblock}[2]{%
  \pic[shift={(-2.2, #2, 0)}] at (0,0,0)
      {Box={name=embed_#1, caption=, xlabel={{,}}, zlabel={},
            fill=EmbedColor, height=12, width=2.5, depth=12}};
  \pic[shift={(0.4, #2, 0)}] at (0,0,0)
      {RightBandedBox={name=nca_#1, caption=, xlabel={{,,}}, zlabel={},
            fill=NCAColor, bandfill=NCABandColor,
            height=12, width={2.5,2.5,2.5}, depth=12}};
  \pic[shift={(4.0, #2, 0)}] at (0,0,0)
      {Box={name=readout_#1, caption=, xlabel={{,}}, zlabel={},
            fill=ReadoutColor, height=12, width=2.5, depth=12}};
  \draw[connection, -Stealth] (embed_#1-east) -- (nca_#1-west);
  \draw[connection, -Stealth] (nca_#1-east)   -- (readout_#1-west);
  % loop arrow under the body
  \path let \p1 = (nca_#1-east), \p2 = (nca_#1-west) in
    coordinate (loopOutR_#1)  at (\x1+0.4cm, \y1)
    coordinate (loopOutRD_#1) at (\x1+0.4cm, \y1-1.7cm)
    coordinate (loopInLD_#1)  at (\x2-0.6cm, \y2-1.7cm)
    coordinate (loopInL_#1)   at (\x2-0.6cm, \y2);
  \draw[-Stealth, line width=0.5mm, opacity=0.32, rounded corners=6pt]
      (nca_#1-east) -- (loopOutR_#1) -- (loopOutRD_#1)
      -- (loopInLD_#1) -- (loopInL_#1) -- (nca_#1-west);
}
