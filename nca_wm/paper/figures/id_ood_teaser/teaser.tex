% ID/OOD teaser figure for the rule-conditioned NCA WM paper.
%
% Single-document standalone TikZ figure.
%
% Per game row: a "card" pair — a code Box flush with the back of the
% PuzzleScript state Box (sharing the back/far face). The state image
% is projected on the *front* (near, z+) face of the state box; the
% code text is projected on the *back* (near face of the code box,
% which sticks out behind the state box because the code box is
% positioned slightly back in z and slightly larger in y so it peeks
% out as a "halo" above/below the state box). The code box feeds into
% the Encoder of the corresponding NCA-WM block; the state feeds into
% the Embed.
%
% Each NCA-WM section additionally includes a small PCA scatter of the
% encoder's latent space next to the Encoder block: train games as
% gray dots; the OOD half adds the held-out games as red stars
% (rendered separately by render_latent_scatters.py).
%

% --- PS frame: small Box at z=0 with image projected on east face. ---
\newcommand{\PSframe}[4]{%
  \pic[shift={(#2, #3, 0)}] at (0,0,0)
      {Box={name=#1, caption=, xlabel={{,}}, zlabel={},
            fill=PSFrameFill, height=4, width=1.0, depth=4}};%
  \node[canvas is zy plane at x={#2 + 0.2}] at (0, #3)
      {\includegraphics[width=0.8cm,height=0.8cm]{#4}};%
}

% --- "Card pair": code Box on the LEFT of the state Box in the
%     reading direction (not depth-wise behind). Code text is
%     projected on the code box's east face — the same plane as the
%     state box's west face. Code box is offset slightly back in z so
%     its east face peeks out behind the state box in iso projection.
%
%     Anchors: <name>g (state box) and <name>c (code box).
%     #1 = name base, #2 = x, #3 = y,
%     #4 = state image path, #5 = code text body
\newcommand{\StateOnCode}[5]{%
  % Code box: glued to the LEFT of the state box (state-west = code-east),
  % same depth, slightly larger height so the top edge of the code box
  % peeks above the state box and the east face has visible area there.
  \pic[shift={({#2 - 0.2}, #3, 0)}] at (0,0,0)
      {Box={name=#1c, caption=, xlabel={{,}}, zlabel={},
            fill=CodeFill, opacity=1.0,
            height=5, width=1.0, depth=4}};%
  % Code text on the east face of the code box, offset forward in z so
  % it lands in the visible front portion of that face (the front-strip
  % that the state box doesn't occlude in iso projection).
  \node[canvas is zy plane at x={#2}, font=\tiny\ttfamily,
        align=left, inner sep=0.5pt, text width=1.0cm,
        anchor=center]
        at (-0.6, #3) {#5};%
  % state box on the right
  \pic[shift={(#2, #3, 0)}] at (0,0,0)
      {Box={name=#1g, caption=, xlabel={{,}}, zlabel={},
            fill=PSFrameFill, height=4, width=1.0, depth=4}};%
  \node[canvas is zy plane at x={#2 + 0.2}] at (0, #3)
      {\includegraphics[width=0.8cm,height=0.8cm]{#4}};%
}

% --- NCA-WM block: Encoder + Embed + 3 NCA bands + Readout, plus a
%     faint loop arrow under the body. The Encoder is a small banded
%     block reminiscent of nca_arch_v1.tex's perceiver encoder.
\newcommand{\NCAblock}[3]{%
  % Encoder: a small banded box (mimics 2 self-attn + 1 cross-attn).
  \pic[shift={({#2 - 2.6}, #3, 0)}] at (0,0,0)
      {RightBandedBox={name=enc_#1, caption=, xlabel={{,,}}, zlabel={},
            fill=EncColor, bandfill=EncBandColor,
            height=8, width={0.45,0.45,0.45}, depth=8}};
  % Embed
  \pic[shift={({#2 - 1.0}, #3, 0)}] at (0,0,0)
      {Box={name=embed_#1, caption=, xlabel={{,}}, zlabel={},
            fill=EmbedColor, height=8, width=1.5, depth=8}};
  % NCA bands
  \pic[shift={({#2 + 0.4}, #3, 0)}] at (0,0,0)
      {RightBandedBox={name=nca_#1, caption=, xlabel={{,,}}, zlabel={},
            fill=NCAColor, bandfill=NCABandColor,
            height=8, width={1.5,1.5,1.5}, depth=8}};
  % Readout
  \pic[shift={({#2 + 2.1}, #3, 0)}] at (0,0,0)
      {Box={name=readout_#1, caption=, xlabel={{,}}, zlabel={},
            fill=ReadoutColor, height=8, width=1.5, depth=8}};
  % Forward path through the dynamics row
  \draw[connection, -Stealth] (embed_#1-east) -- (nca_#1-west);
  \draw[connection, -Stealth] (nca_#1-east)   -- (readout_#1-west);
  % Encoder → NCA bands top (slot conditioning)
  \draw[condarrow] (enc_#1-east)
      to[out=20, in=120]
      ($(nca_#1-north)+(0,0)$);
  % Iteration loop arrow under the body
  \path let \p1 = (nca_#1-east), \p2 = (nca_#1-west) in
    coordinate (loopOutR_#1)  at (\x1+0.25cm, \y1)
    coordinate (loopOutRD_#1) at (\x1+0.25cm, \y1-1.0cm)
    coordinate (loopInLD_#1)  at (\x2-0.35cm, \y2-1.0cm)
    coordinate (loopInL_#1)   at (\x2-0.35cm, \y2);
  \draw[-Stealth, line width=0.4mm, opacity=0.32, rounded corners=4pt]
      (nca_#1-east) -- (loopOutR_#1) -- (loopOutRD_#1)
      -- (loopInLD_#1) -- (loopInL_#1) -- (nca_#1-west);
}
