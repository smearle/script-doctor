% ID/OOD teaser figure for the rule-conditioned NCA WM paper.
%
% Single-document standalone TikZ figure.
%
% Per game row: a transparent code Box flush against the WEST face of
% a transparent PuzzleScript state Box. Code on the shared east face,
% state image on the state's east face.
%
% Per section: a single skinny Encoder (one blue block) feeds a yellow
% Latent block ($z$). The latent block conditions every NCA application
% (drawn as just the 3 green NCA bands). ID uses ONE NCA (vertically
% centered between rows) between s_0 and s_1; OOD uses ONE NCA per
% timestep, also vertically centered, with all rows fanning in/out.
%
% Latent scatters (large, above the figure) are connected from the
% yellow latent block by a dashed line.
%

% --- PS state frame (transparent), bigger than before. ---
\newcommand{\PSframe}[4]{%
  \pic[shift={(#2, #3, 0)}] at (0,0,0)
      {Box={name=#1, caption=, xlabel={{,}}, zlabel={},
            fill=PSFrameFill, opacity=0.35,
            height=8, width=1.0, depth=8}};%
  \node[canvas is zy plane at x={#2 + 0.2}] at (0, #3)
      {\includegraphics[width=1.6cm,height=1.6cm]{#4}};%
}

% --- Card pair: code box LEFT of state box, both transparent.
%     Bigger than before so the code text is more legible.
\newcommand{\StateOnCode}[5]{%
  \pic[shift={({#2 - 0.2}, #3, 0)}] at (0,0,0)
      {Box={name=#1c, caption=, xlabel={{,}}, zlabel={},
            fill=CodeFill, opacity=0.35,
            height=8, width=1.0, depth=8}};%
  \node[canvas is zy plane at x={#2}, font=\scriptsize\ttfamily,
        align=left, inner sep=0pt, text width=1.5cm,
        anchor=center]
        at (0, #3) {#5};%
  \pic[shift={(#2, #3, 0)}] at (0,0,0)
      {Box={name=#1g, caption=, xlabel={{,}}, zlabel={},
            fill=PSFrameFill, opacity=0.35,
            height=8, width=1.0, depth=8}};%
  \node[canvas is zy plane at x={#2 + 0.2}] at (0, #3)
      {\includegraphics[width=1.6cm,height=1.6cm]{#4}};%
}

% --- NCA "step": just the 3 green bands at (x, y).
\newcommand{\NCAstep}[3]{%
  \pic[shift={(#2, #3, 0)}] at (0,0,0)
      {RightBandedBox={name=n#1, caption=, xlabel={{,,}}, zlabel={},
            fill=NCAColor, bandfill=NCABandColor,
            height=4, width={1.0,1.0,1.0}, depth=4}};%
}

% --- Encoder + Latent: one skinny blue Encoder + a yellow Latent ($z$)
%     block to its right.
\newcommand{\EncLatent}[3]{%
  \node[draw, rounded corners=1.5pt, fill=EncColor, line width=0.4pt,
        minimum width=0.30cm, minimum height=0.80cm, anchor=center]
        (enc_#1) at (#2, #3) {};
  \node[draw, rounded corners=1.5pt, fill=SlotColor, line width=0.4pt,
        minimum width=0.32cm, minimum height=0.80cm, anchor=west,
        font=\tiny] (slots_#1) at ({#2 + 0.45}, #3) {$z$};
  \draw[arrow, color=black!50] (enc_#1.east) -- (slots_#1.west);
}
